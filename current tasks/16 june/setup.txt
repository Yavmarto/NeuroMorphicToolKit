#!/bin/bash
set -e

echo "Starting VM setup..."



# 2. APT Packages Installed
echo "Installing APT packages..."
sudo apt update
sudo apt install -y git curl wget build-essential unzip tar ca-certificates tmux rsync bash file xz-utils zip apt-transport-https gnupg clang cmake ninja-build pkg-config libgtk-3-dev liblzma-dev libstdc++6 libglu1-mesa default-jdk android-tools-adb nodejs gh

# 3. Android SDK Setup
echo "Setting up Android SDK..."
mkdir -p ~/android-sdk/cmdline-tools
cd ~/android-sdk/cmdline-tools
wget https://dl.google.com/android/repository/commandlinetools-linux-11076708_latest.zip
unzip commandlinetools-linux-*_latest.zip
rm commandlinetools-linux-*_latest.zip
mv cmdline-tools latest

# Accept licenses and install platform tools
cd latest/bin
yes | ./sdkmanager --licenses
./sdkmanager "platform-tools" "build-tools;34.0.0" "platforms;android-34"

# Use system-native ARM adb instead of the bundled intel one
rm -f ~/android-sdk/platform-tools/adb
ln -s /usr/bin/adb ~/android-sdk/platform-tools/adb

# 4. Miniforge Setup
echo "Installing Miniforge..."
cd ~
wget https://github.com/conda-forge/miniforge/releases/latest/download/Miniforge3-Linux-aarch64.sh
bash Miniforge3-Linux-aarch64.sh -b -p $HOME/miniforge3
~/miniforge3/bin/conda init bash

# 5. Flutter Installation & Configuration
echo "Installing and Configuring Flutter..."

# Create the directory and grant your user ownership so flutter can update itself
sudo mkdir -p /opt/flutter
sudo chown -R $USER:$USER /opt/flutter

# Clone the latest stable release of Flutter
git clone https://github.com/flutter/flutter.git -b stable /opt/flutter

# Temporarily add flutter to PATH for this script session
export PATH="$PATH:/opt/flutter/bin"

# Configure Flutter
flutter config --no-enable-linux-desktop || true
flutter config --android-sdk ~/android-sdk || true

# Pre-download development binaries and check setup
flutter precache || true
flutter doctor -v || true

# 6. Bash Environment Variables
echo "Updating ~/.bashrc..."
grep -q "JAVA_HOME=/usr/lib/jvm/default-java" ~/.bashrc || echo 'export JAVA_HOME=/usr/lib/jvm/default-java' >> ~/.bashrc
grep -q "ANDROID_HOME=\$HOME/android-sdk" ~/.bashrc || echo 'export ANDROID_HOME=$HOME/android-sdk' >> ~/.bashrc
grep -q "ANDROID_HOME/cmdline-tools" ~/.bashrc || echo 'export PATH=$PATH:$JAVA_HOME/bin:$ANDROID_HOME/cmdline-tools/latest/bin:$ANDROID_HOME/platform-tools' >> ~/.bashrc
grep -q "/opt/flutter/bin" ~/.bashrc || echo 'export PATH="$PATH:/opt/flutter/bin"' >> ~/.bashrc

# 7. Cloned Projects
echo "Cloning projects..."
cd ~
git clone --recursive -b dev https://Yavmarto@github.com/Completed-Spoon-6/NeuroMorphicToolKit.git || echo "Repo might already exist."

echo "Setup complete! Please run 'source ~/.bashrc' or restart your terminal."
